import datetime
from dateutil.relativedelta import relativedelta

print("Birthday Calculator")
print("Current Day")
currentMonth = int(input("Month: "))
currentDay = int(input("Day: "))
currentYear = int(input("Year: "))

currentDay = datetime.date(currentYear, currentMonth, currentDay)
print("Birthday")
birthdayMonth = int(input("Month: "))
birthdayDay = int(input("Day: "))
birthdayYear = int(input("Year: "))
birthday = datetime.date(birthdayYear, birthdayMonth, birthdayDay)

print(relativedelta(birthday, currentDay).years)
